   308→function AnimatedCounter({ target, duration = 2, suffix = '' }: { target: number, duration?: number, suffix?: string }) {
   309→  const [count, setCount] = useState(0)
   310→  const ref = useRef<HTMLSpanElement>(null)
   311→  const isInView = useInView(ref, { once: true, margin: '-100px' })
   312→
   313→  useEffect(() => {
   314→    if (!isInView) return
   315→    let start = 0
   316→    const increment = target / (duration * 60)
   317→    const timer = setInterval(() => {
   318→      start += increment
   319→      if (start >= target) {
   320→        setCount(target)
   321→        clearInterval(timer)
   322→      } else {
   323→        setCount(Math.floor(start))
   324→      }
   325→    }, 1000 / 60)
   326→    return () => clearInterval(timer)
   327→  }, [isInView, target, duration])
